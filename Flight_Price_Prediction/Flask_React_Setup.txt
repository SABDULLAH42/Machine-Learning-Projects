import { useState } from "react";

function App() {
  const [formData, setFormData] = useState({
    airline: "",
    source_city: "",
    departure_time: "",
    stops: "",
    arrival_time: "",
    destination_city: "",
    class: "",
    departure_date: "",
  });

  const [prediction, setPrediction] = useState(null);
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  // Dropdown options
  const airlines = [
    "AirAsia",
    "Indigo",
    "GO_FIRST",
    "SpiceJet",
    "Air_India",
    "Vistara",
  ];

  const cities = [
    "Delhi",
    "Hyderabad",
    "Bangalore",
    "Mumbai",
    "Kolkata",
    "Chennai",
  ];

  const departureTimes = [
    "Early_Morning",
    "Morning",
    "Afternoon",
    "Evening",
    "Night",
    "Late_Night",
  ];

  const stops = [
    "zero",
    "one",
    "two_or_more",
  ];

  const arrivalTimes = [
    "Early_Morning",
    "Morning",
    "Afternoon",
    "Evening",
    "Night",
    "Late_Night",
  ];

  const classes = [
    "Economy",
    "Business",
  ];

  // Handle input changes
  const handleChange = (event) => {
    const { name, value } = event.target;

    setFormData({
      ...formData,
      [name]: value,
    });
  };

  // Send data to Flask
  const handleSubmit = async (event) => {
    event.preventDefault();

    setPrediction(null);
    setError("");
    setLoading(true);

    try {
      const response = await fetch(
        "http://127.0.0.1:5000/predict",
        {
          method: "POST",

          headers: {
            "Content-Type": "application/json",
          },

          body: JSON.stringify(formData),
        }
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.error || "Prediction failed");
      }

      setPrediction(data.prediction);

    } catch (error) {
      setError(
        error.message ||
        "Unable to connect to the prediction server."
      );
    } finally {
      setLoading(false);
    }
  };

  // Reset form
  const handleReset = () => {
    setFormData({
      airline: "",
      source_city: "",
      departure_time: "",
      stops: "",
      arrival_time: "",
      destination_city: "",
      class: "",
      departure_date: "",
    });

    setPrediction(null);
    setError("");
  };

  return (
    <div className="min-h-screen bg-gray-100 py-10 px-4">

      {/* Main Container */}
      <div className="max-w-5xl mx-auto">

        {/* Header */}
        <div className="text-center mb-10">

          <h1 className="text-4xl font-bold text-blue-700">
            Flight Price Prediction
          </h1>

          <p className="text-gray-600 mt-3">
            Enter your flight details to predict the ticket price
          </p>

        </div>


        {/* Form Card */}
        <div className="bg-white rounded-2xl shadow-lg p-8">

          <form onSubmit={handleSubmit}>

            {/* Form Grid */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">


              {/* Airline */}
              <div>
                <label className="block text-sm font-semibold text-gray-700 mb-2">
                  Airline
                </label>

                <select
                  name="airline"
                  value={formData.airline}
                  onChange={handleChange}
                  required
                  className="w-full border border-gray-300 rounded-lg px-4 py-3 focus:outline-none focus:ring-2 focus:ring-blue-500"
                >

                  <option value="">
                    Select Airline
                  </option>

                  {airlines.map((airline) => (
                    <option
                      key={airline}
                      value={airline}
                    >
                      {airline}
                    </option>
                  ))}

                </select>
              </div>


              {/* Source City */}
              <div>
                <label className="block text-sm font-semibold text-gray-700 mb-2">
                  Source City
                </label>

                <select
                  name="source_city"
                  value={formData.source_city}
                  onChange={handleChange}
                  required
                  className="w-full border border-gray-300 rounded-lg px-4 py-3 focus:outline-none focus:ring-2 focus:ring-blue-500"
                >

                  <option value="">
                    Select Source City
                  </option>

                  {cities.map((city) => (
                    <option
                      key={city}
                      value={city}
                    >
                      {city}
                    </option>
                  ))}

                </select>
              </div>


              {/* Departure Time */}
              <div>
                <label className="block text-sm font-semibold text-gray-700 mb-2">
                  Departure Time
                </label>

                <select
                  name="departure_time"
                  value={formData.departure_time}
                  onChange={handleChange}
                  required
                  className="w-full border border-gray-300 rounded-lg px-4 py-3 focus:outline-none focus:ring-2 focus:ring-blue-500"
                >

                  <option value="">
                    Select Departure Time
                  </option>

                  {departureTimes.map((time) => (
                    <option
                      key={time}
                      value={time}
                    >
                      {time.replace("_", " ")}
                    </option>
                  ))}

                </select>
              </div>


              {/* Stops */}
              <div>
                <label className="block text-sm font-semibold text-gray-700 mb-2">
                  Number of Stops
                </label>

                <select
                  name="stops"
                  value={formData.stops}
                  onChange={handleChange}
                  required
                  className="w-full border border-gray-300 rounded-lg px-4 py-3 focus:outline-none focus:ring-2 focus:ring-blue-500"
                >

                  <option value="">
                    Select Stops
                  </option>

                  {stops.map((stop) => (
                    <option
                      key={stop}
                      value={stop}
                    >
                      {stop === "zero"
                        ? "Non-Stop"
                        : stop === "one"
                        ? "1 Stop"
                        : "2 or More Stops"}
                    </option>
                  ))}

                </select>
              </div>


              {/* Arrival Time */}
              <div>
                <label className="block text-sm font-semibold text-gray-700 mb-2">
                  Arrival Time
                </label>

                <select
                  name="arrival_time"
                  value={formData.arrival_time}
                  onChange={handleChange}
                  required
                  className="w-full border border-gray-300 rounded-lg px-4 py-3 focus:outline-none focus:ring-2 focus:ring-blue-500"
                >

                  <option value="">
                    Select Arrival Time
                  </option>

                  {arrivalTimes.map((time) => (
                    <option
                      key={time}
                      value={time}
                    >
                      {time.replace("_", " ")}
                    </option>
                  ))}

                </select>
              </div>


              {/* Destination City */}
              <div>
                <label className="block text-sm font-semibold text-gray-700 mb-2">
                  Destination City
                </label>

                <select
                  name="destination_city"
                  value={formData.destination_city}
                  onChange={handleChange}
                  required
                  className="w-full border border-gray-300 rounded-lg px-4 py-3 focus:outline-none focus:ring-2 focus:ring-blue-500"
                >

                  <option value="">
                    Select Destination City
                  </option>

                  {cities.map((city) => (
                    <option
                      key={city}
                      value={city}
                    >
                      {city}
                    </option>
                  ))}

                </select>
              </div>


              {/* Class */}
              <div>
                <label className="block text-sm font-semibold text-gray-700 mb-2">
                  Class
                </label>

                <select
                  name="class"
                  value={formData.class}
                  onChange={handleChange}
                  required
                  className="w-full border border-gray-300 rounded-lg px-4 py-3 focus:outline-none focus:ring-2 focus:ring-blue-500"
                >

                  <option value="">
                    Select Class
                  </option>

                  {classes.map((flightClass) => (
                    <option
                      key={flightClass}
                      value={flightClass}
                    >
                      {flightClass}
                    </option>
                  ))}

                </select>
              </div>


              {/* Departure Date */}
              <div>
                <label className="block text-sm font-semibold text-gray-700 mb-2">
                  Departure Date
                </label>

                <input
                  type="date"
                  name="departure_date"
                  value={formData.departure_date}
                  onChange={handleChange}
                  required
                  className="w-full border border-gray-300 rounded-lg px-4 py-3 focus:outline-none focus:ring-2 focus:ring-blue-500"
                />

              </div>

            </div>


            {/* Buttons */}
            <div className="flex flex-col sm:flex-row gap-4 mt-8">

              <button
                type="submit"
                disabled={loading}
                className="flex-1 bg-blue-600 text-white py-3 rounded-lg font-semibold hover:bg-blue-700 transition disabled:bg-blue-300"
              >

                {loading
                  ? "Predicting..."
                  : "Predict Flight Price"}

              </button>


              <button
                type="button"
                onClick={handleReset}
                className="sm:w-40 bg-gray-200 text-gray-700 py-3 rounded-lg font-semibold hover:bg-gray-300 transition"
              >
                Reset
              </button>

            </div>

          </form>


          {/* Prediction Result */}
          {prediction !== null && (

            <div className="mt-8 bg-green-50 border border-green-200 rounded-xl p-6 text-center">

              <p className="text-gray-600 text-lg">
                Estimated Flight Price
              </p>

              <h2 className="text-4xl font-bold text-green-600 mt-2">
                ₹{Number(prediction).toLocaleString("en-IN")}
              </h2>

            </div>

          )}


          {/* Error */}
          {error && (

            <div className="mt-6 bg-red-50 border border-red-200 rounded-xl p-4">

              <p className="text-red-600 text-center font-medium">
                {error}
              </p>

            </div>

          )}

        </div>


        {/* Footer */}
        <p className="text-center text-gray-500 text-sm mt-6">
          Flight Price Prediction using Machine Learning
        </p>

      </div>

    </div>
  );
}

export default App;
